// 1. Declaraciones Globales
int contador = 0;
boolean buscando = true;
int lectura = 0;
int temp = 0; // Variable temporal para el bucle de espera

// 2. Definicion de Setup
void setup() {
    
    // ============================
    // FASE 1: AVANZAR Y CONTAR 4
    // ============================
    contador = 0;
    buscando = true;
    
    while (buscando) {
        
        mover();
        lectura = leerSensor();
        
        // Si detectamos linea negra (ambos sensores sobre la línea)
        if (lectura == 1) {
            
            // 1. Contamos ESTA linea
            contador = contador + 1;
            
            // 2. Esperamos a salir de la línea para no contarla dos veces
            temp = 1; 
            while (temp == 1) {
                mover();
                temp = leerSensor();
            }
        }
        
        // Verificar si terminamos las 4 líneas hacia adelante
        if (contador >= 4) {
            buscando = false;
        }
    }

    parar();
    reversa(170, 40);

    // ============================
    // FASE 2: REVERSA Y CONTAR 4
    // ============================
    contador = 0;
    buscando = true;

    while (buscando) {
        
        reversa();
        lectura = leerSensor();
        
        // Si detectamos linea negra en reversa
        if (lectura == 1) {
            
            // 1. Contamos ESTA linea
            contador = contador + 1;
            
            // 2. Esperamos a salir de la línea para no contarla dos veces
            temp = 1;
            while (temp == 1) {
                reversa();
                temp = leerSensor();
            }
        }
        
        // Verificar si terminamos las 4 líneas en reversa
        if (contador >= 4) {
            buscando = false;
        }
    }

    // Freno final completo
    parar();
}
