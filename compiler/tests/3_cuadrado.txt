// 1. Declaraciones Globales
int velLado = 200;     // velocidad PWM para los lados del cuadrado
int tLado = 1200;      // tiempo (ms) para avanzar un lado
int tGiro = 1700;       // tiempo (ms) para un giro ~90° (ajusta según tu robot)

// 2. Definicion de Setup
void setup() {
    while (true) {
        // Avanzar un lado
        mover(velLado, tLado);   // avanza a velLado durante tLado ms
        
        // Girar 90 grados (aprox) hacia la derecha
        girarDer(tGiro);         // gira durante tGiro ms
    }
}
