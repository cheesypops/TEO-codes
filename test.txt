// ROBOT SEGUIDOR DE LINEA (Gramatica Propia)

// 1. CONFIGURACION GLOBAL
int velocidad_base = 200;   // Velocidad para avanzar recto
int velocidad_giro = 180;   // Velocidad para corregir trayectoria
int demora = 20;            // Pausa en ms para estabilidad

// 2. FUNCION DE CONTROL DE MOTORES
// Recibe el estado de los sensores y decide que hacer.
// 0 = Blanco / Blanco -> Linea centrada -> AVANZAR
// 1 = Blanco / Negro  -> Linea a la derecha -> GIRAR DERECHA
// 2 = Negro  / Blanco -> Linea a la izquierda -> GIRAR IZQUIERDA
// 3 = Negro  / Negro  -> Cruce o Final -> PARAR

void controlar_movimiento(int estado) {
    
    if (estado == 0) {
        // Caso: Todo Blanco (Avanzar)
        token_mover(velocidad_base);
    } 
    else {
        // Bloques anidados para simular else if
        if (estado == 1) {
            // Caso: Derecha pisa Negro (Corregir a Derecha)
            token_girarDer(velocidad_giro);
        } 
        else {
            if (estado == 2) {
                // Caso: Izquierda pisa Negro (Corregir a Izquierda)
                token_girarIzq(velocidad_giro);
            } 
            else {
                // Caso: Ambos Negro (Detenerse)
                token_parar(0);
            }
        }
    }
}

// 3. FUNCION PRINCIPAL (Entry Point)
void main() {
    int lectura = 0;

    // Ciclo infinito de control
    while (true) {
        
        // A. Leer valor de los sensores IR
        lectura = token_leerSensor();

        // B. Llamar a la logica de decision
        controlar_movimiento(lectura);

        // C. Pequena espera
        token_esperar(demora);
    }
}
